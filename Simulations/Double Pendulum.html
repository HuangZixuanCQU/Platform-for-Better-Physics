import matplotlib.pyplot as plt import matplotlib.animation as animation import numpy as np from pyodide.ffi import create_proxy from js import document from matplotlib.patches import Circle # ---------------------------------------------------------------- # DOUBLE PENDULUM PHYSICS # ---------------------------------------------------------------- g = 9.8 # gravity (m/s²) def derivatives(state, L1, L2, m1, m2): """Compute derivatives for double pendulum using Lagrangian mechanics.""" theta1, theta2, omega1, omega2 = state dtheta1 = omega1 dtheta2 = omega2 # Compute angular accelerations delta = theta2 - theta1 den1 = (m1 + m2) * L1 - m2 * L1 * np.cos(delta) * np.cos(delta) domega1 = (m2 * L1 * omega1 * omega1 * np.sin(delta) * np.cos(delta) + m2 * L2 * omega2 * omega2 * np.sin(delta) + (m1 + m2) * g * np.sin(theta1) * np.cos(delta) - (m1 + m2) * g * np.sin(theta2)) / den1 den2 = (L2 / L1) * den1 domega2 = (-m2 * L2 * omega2 * omega2 * np.sin(delta) * np.cos(delta) + (m1 + m2) * g * np.sin(theta1) * np.cos(delta) - (m1 + m2) * g * np.sin(theta2) - (m1 + m2) * L1 * omega1 * omega1 * np.sin(delta)) / den2 return np.array([dtheta1, dtheta2, domega1, domega2]) def simulate_pendulum(L1, L2, m1, m2, dt=0.01, steps=500): """Run simulation and return trajectory.""" # Initial conditions: start from rest with small displacement state = np.array([np.radians(30), np.radians(-30), 0.0, 0.0]) times = np.linspace(0, steps * dt, steps) states = np.zeros((steps, 4)) states[0] = state for i in range(1, steps): k1 = derivatives(state, L1, L2, m1, m2) k2 = derivatives(state + 0.5 * dt * k1, L1, L2, m1, m2) k3 = derivatives(state + 0.5 * dt * k2, L1, L2, m1, m2) k4 = derivatives(state + dt * k3, L1, L2, m1, m2) state = state + (dt / 6.0) * (k1 + 2 * k2 + 2 * k3 + k4) states[i] = state return times, states def compute_energy(theta1, theta2, omega1, omega2, L1, L2, m1, m2): """Compute total mechanical energy.""" # Kinetic energy v1x = -L1 * omega1 * np.sin(theta1) v1y = L1 * omega1 * np.cos(theta1) v2x = v1x - L2 * omega2 * np.sin(theta2) v2y = v1y + L2 * omega2 * np.cos(theta2) KE = 0.5 * m1 * (v1x**2 + v1y**2) + 0.5 * m2 * (v2x**2 + v2y**2) # Potential energy (zero at pivot) PE = -m1 * g * L1 * np.cos(theta1) - m2 * g * (L1 * np.cos(theta1) + L2 * np.cos(theta2)) return KE + PE # ---------------------------------------------------------------- # PLOTTING # ---------------------------------------------------------------- def plot_pendulum(L1, L2, m1, m2): times, states = simulate_pendulum(L1, L2, m1, m2) theta1 = states[:, 0] theta2 = states[:, 1] omega1 = states[:, 2] omega2 = states[:, 3] # Positions x1 = L1 * np.sin(theta1) y1 = -L1 * np.cos(theta1) x2 = x1 + L2 * np.sin(theta2) y2 = y1 - L2 * np.cos(theta2) # Energy energy = np.array([compute_energy(t1, t2, w1, w2, L1, L2, m1, m2) for t1, t2, w1, w2 in zip(theta1, theta2, omega1, omega2)]) # Create figure with two subplots fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(10, 5)) # ---- Left: Pendulum animation (final frame) ---- ax1.set_xlim(-(L1 + L2 + 20), L1 + L2 + 20) ax1.set_ylim(-(L1 + L2 + 20), L1 + L2 + 20) ax1.set_aspect('equal') ax1.grid(True, alpha=0.3) ax1.set_xlabel('x (px)') ax1.set_ylabel('y (px)') ax1.set_title('Double Pendulum') # Draw pendulum at final state idx = -1 # Rod 1 ax1.plot([0, x1[idx]], [0, y1[idx]], 'k-', linewidth=2) # Rod 2 ax1.plot([x1[idx], x2[idx]], [y1[idx], y2[idx]], 'k-', linewidth=2) # Pivot ax1.plot(0, 0, 'ko', markersize=8) # Mass 1 ax1.add_patch(Circle((x1[idx], y1[idx]), radius=0.15 * max(L1, L2), color='#2d7aff', alpha=0.8)) # Mass 2 ax1.add_patch(Circle((x2[idx], y2[idx]), radius=0.12 * max(L1, L2), color='#ff6b35', alpha=0.8)) # Trail ax1.plot(x1, y1, color='#2d7aff', linewidth=1, alpha=0.3) ax1.plot(x2, y2, color='#ff6b35', linewidth=1, alpha=0.3) # ---- Right: Energy plot ---- ax2.plot(times, energy, color='#28a745', linewidth=2) ax2.set_xlabel('Time (s)') ax2.set_ylabel('Energy (J)') ax2.set_title('Total Mechanical Energy') ax2.grid(True, alpha=0.3) ax2.axhline(y=energy[0], color='red', linestyle='--', alpha=0.5, label=f'Initial E = {energy[0]:.2f} J') ax2.legend(fontsize=10) # Energy conservation check energy_drift = (np.max(energy) - np.min(energy)) / np.mean(energy) * 100 ax2.text(0.05, 0.95, f'Energy drift: {energy_drift:.2f}%', transform=ax2.transAxes, fontsize=10, verticalalignment='top', bbox=dict(boxstyle='round', facecolor='white', alpha=0.8)) plt.tight_layout() # Update info panel doc = document doc.getElementById('energyDisplay').innerHTML = f'{energy[-1]:.2f} J' doc.getElementById('theta1Display').innerHTML = f'{np.degrees(theta1[-1]):.1f} °' doc.getElementById('theta2Display').innerHTML = f'{np.degrees(theta2[-1]):.1f} °' doc.getElementById('timeDisplay').innerHTML = f'{times[-1]:.2f} s' return fig # ---------------------------------------------------------------- # UI BINDINGS # ---------------------------------------------------------------- L1_slider = document.getElementById('L1Slider') L2_slider = document.getElementById('L2Slider') m1_slider = document.getElementById('m1Slider') m2_slider = document.getElementById('m2Slider') def update_plot(event=None): L1 = float(L1_slider.value) L2 = float(L2_slider.value) m1 = float(m1_slider.value) m2 = float(m2_slider.value) container = document.getElementById('plot-container') container.innerHTML = '
