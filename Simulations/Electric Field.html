from pyscript import display import matplotlib.pyplot as plt import numpy as np from pyodide.ffi import create_proxy from js import document # ---------------------------------------------------------------- # PHYSICS: Electric field of point charges # ---------------------------------------------------------------- k = 1.0 # Coulomb constant (scaled for visualisation) def electric_field(q, pos, x, y): """Compute electric field at (x, y) due to charge q at position pos.""" dx = x - pos[0] dy = y - pos[1] r2 = dx*dx + dy*dy if r2 < 0.01: return 0, 0 r = np.sqrt(r2) E = k * q / r2 return E * dx / r, E * dy / r def total_field(charges, x, y): """Compute total electric field at (x, y) from all charges.""" Ex = np.zeros_like(x) Ey = np.zeros_like(y) for q, pos in charges: ex, ey = electric_field(q, pos, x, y) Ex += ex Ey += ey return Ex, Ey def total_potential(charges, x, y): """Compute total electric potential at (x, y) from all charges.""" V = np.zeros_like(x) for q, pos in charges: dx = x - pos[0] dy = y - pos[1] r = np.sqrt(dx*dx + dy*dy) # Avoid division by zero: set minimum distance r = np.maximum(r, 0.1) V += k * q / r return V # ---------------------------------------------------------------- # FIELD LINE TRACING # ---------------------------------------------------------------- def trace_field_line(charges, start_x, start_y, step=0.05, max_steps=500): """Trace a single field line from a starting point.""" x, y = start_x, start_y points_x = [x] points_y = [y] for _ in range(max_steps): Ex, Ey = total_field(charges, x, y) E_mag = np.sqrt(Ex*Ex + Ey*Ey) if E_mag < 1e-8: break dx = step * Ex / E_mag dy = step * Ey / E_mag x += dx y += dy points_x.append(x) points_y.append(y) for q, pos in charges: if np.sqrt((x - pos[0])**2 + (y - pos[1])**2) < 0.2: return np.array(points_x), np.array(points_y) if abs(x) > 8 or abs(y) > 8: break return np.array(points_x), np.array(points_y) def get_starting_points(charges, n_lines=24): """Generate starting points around positive charges.""" start_points = [] for q, pos in charges: if q > 0: for i in range(n_lines): angle = 2 * np.pi * i / n_lines r = 0.3 start_points.append((pos[0] + r * np.cos(angle), pos[1] + r * np.sin(angle))) return start_points # Grid for potential grid_size = 200 x = np.linspace(-8, 8, grid_size) y = np.linspace(-8, 8, grid_size) X, Y = np.meshgrid(x, y) # Compute potential - safe version V = total_potential(charges, X, Y) # Replace inf with finite values V = np.nan_to_num(V, nan=0.0, posinf=15.0, neginf=-15.0) # Equipotential contours levels = np.linspace(-12, 12, 20) ax.contour(X, Y, V, levels=levels, colors='#dc3545', alpha=0.5, linewidths=0.8) #
