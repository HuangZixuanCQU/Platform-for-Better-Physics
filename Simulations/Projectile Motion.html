import matplotlib.pyplot as plt import numpy as np from pyodide.ffi import create_proxy from js import document # ---------------------------------------------------------------- # PHYSICS # ---------------------------------------------------------------- def get_range(v0, theta_deg, g): theta = np.radians(theta_deg) return (v0**2 * np.sin(2 * theta)) / g def get_max_height(v0, theta_deg, g): theta = np.radians(theta_deg) return (v0**2 * np.sin(theta)**2) / (2 * g) def get_time_of_flight(v0, theta_deg, g): theta = np.radians(theta_deg) return (2 * v0 * np.sin(theta)) / g def compute_trajectory(v0, theta_deg, g, num_points=300): theta = np.radians(theta_deg) t_flight = get_time_of_flight(v0, theta_deg, g) t = np.linspace(0, t_flight, num_points) x = v0 * np.cos(theta) * t y = v0 * np.sin(theta) * t - 0.5 * g * t**2 y = np.maximum(y, 0) return x, y def compute_safety_parabola(v0, g, num_points=200): max_range = v0**2 / g x = np.linspace(0, max_range, num_points) y = (v0**2) / (2*g) - (g / (2 * v0**2)) * x**2 x_neg = np.linspace(-max_range, 0, num_points) y_neg = (v0**2) / (2*g) - (g / (2 * v0**2)) * x_neg**2 return np.concatenate([x_neg, x]), np.concatenate([y_neg, y]) # ---------------------------------------------------------------- # PLOTTING # ---------------------------------------------------------------- def plot_projectile(v0, theta_deg, g): x, y = compute_trajectory(v0, theta_deg, g) x_safe, y_safe = compute_safety_parabola(v0, g) range_val = get_range(v0, theta_deg, g) max_h = get_max_height(v0, theta_deg, g) fig, ax = plt.subplots(figsize=(10, 6)) ax.set_xlim(-10, max(v0**2/g * 1.15, 100)) ax.set_ylim(-5, max(v0**2/(2*g) * 1.3, 50)) ax.grid(True, linestyle='--', alpha=0.3, color='#cccccc') ax.set_xlabel('Range (m)', fontsize=12) ax.set_ylabel('Height (m)', fontsize=12) ax.set_title(f'Projectile Motion: v₀ = {v0} m/s, θ = {theta_deg}°, g = {g} m/s²', fontsize=14) #
